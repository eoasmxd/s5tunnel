var ipv4Pattern = /^[0-9]{1,3}(\.[0-9]{1,3}){3}$/;
var ipv6LoopbackOrLinkLocalPattern = /^(::1|(0+:){7}0*1)(%.*)?$|^fe[89ab][0-9a-f]:/i;

function isLoopbackHost(host) {
    if (!host) {
        return false;
    }
    var h = host;
    if (h.charAt(0) === '[' && h.charAt(h.length - 1) === ']') {
        h = h.substring(1, h.length - 1);
    }
    h = h.toLowerCase();
    if (h === 'localhost') {
        return true;
    }
    if (ipv4Pattern.test(h) && isInNet(h, '127.0.0.0', '255.0.0.0')) {
        return true;
    }
    if (h.indexOf(':') >= 0 && ipv6LoopbackOrLinkLocalPattern.test(h)) {
        return true;
    }
    return false;
}

function matchNetworks(host, networks) {
    if (!host || !ipv4Pattern.test(host)) {
        return false;
    }
    for (var i = 0; i < networks.length; i++) {
        if (isInNet(host, networks[i][0], networks[i][1])) {
            return true;
        }
    }
    return false;
}

function matchDomains(host, domains) {
    if (!host) {
        return false;
    }
    var h = host.toLowerCase();
    if (h.length > 0 && h.charAt(h.length - 1) === '.') {
        h = h.substring(0, h.length - 1);
    }
    for (var i = 0; i < domains.length; i++) {
        var d = domains[i].toLowerCase();
        if (h === d || dnsDomainIs(h, '.' + d)) {
            return true;
        }
    }
    return false;
}

function FindProxyForURL(url, host) {
    if (isLoopbackHost(host)) {
        return direct;
    }
    if (matchNetworks(host, directNetworks) || matchDomains(host, directDomains)) {
        return direct;
    }
    if (matchNetworks(host, proxyNetworks) || matchDomains(host, proxyDomains)) {
        return proxy;
    }
    return defaultAction;
}
